fun getTheme(mode: ThemeMode): AppColors {

    return when (mode) {

        ThemeMode.OCEAN_LIGHT -> OceanLightTheme

        ThemeMode.OCEAN -> OceanTheme

        ThemeMode.LAVENDER -> LavenderTheme

        ThemeMode.PINK -> PinkTheme

        ThemeMode.FOREST -> ForestTheme

        ThemeMode.NEON -> NeonTheme

        ThemeMode.PURPLE -> PurpleTheme

        ThemeMode.ORANGE -> OrangeTheme

        ThemeMode.ICE -> IceTheme

        ThemeMode.CRIMSON -> CrimsonTheme
        ThemeMode.EARTH -> EarthTheme
        ThemeMode.ROSE -> RoseTheme
        ThemeMode.LAVENDER_SKY -> LavenderSkyTheme
        ThemeMode.OCEAN_GREEN -> OceanGreenTheme
        ThemeMode.FOREST_GREY -> ForestGreyTheme
        ThemeMode.DESERT -> DesertTheme
        ThemeMode.ARCTIC -> ArcticTheme
        ThemeMode.ICE_BLUE -> IceBlueTheme
    }
}
